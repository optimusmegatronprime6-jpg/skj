Recursive Algorithm

function hanoi(n, src, dest, aux):
  if n == 1:
    move(src → dest) // base case
    return

  hanoi(n-1, src, aux, dest)
  move(src → dest) // move largest
  hanoi(n-1, aux, dest, src)
